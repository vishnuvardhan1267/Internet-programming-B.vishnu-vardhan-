<script>

/* USER DATA */

let users=[];
let currentUser="";


/* PRODUCT DATA */

let products=[

{
id:1,
name:"Red Rose Bouquet",
category:"Flowers",
occasion:"Love",
price:899,
desc:"Beautiful bouquet of fresh red roses"
},

{
id:2,
name:"Birthday Flower Basket",
category:"Gift Baskets",
occasion:"Birthday",
price:1499,
desc:"Colorful flowers with birthday gifts"
},

{
id:3,
name:"Chocolate Gift Box",
category:"Birthday",
occasion:"Birthday",
price:799,
desc:"Premium chocolates in a gift box"
},

{
id:4,
name:"Anniversary Roses",
category:"Flowers",
occasion:"Anniversary",
price:1299,
desc:"Special rose arrangement for couples"
},

{
id:5,
name:"Personalized Mug",
category:"Personalized",
occasion:"Love",
price:599,
desc:"Custom printed personalized mug"
}

];

let cart=[];
let orders=[];


/* NAVIGATION */

function show(id){

document.querySelectorAll("section")
.forEach(s=>s.classList.add("hidden"));

document.getElementById(id)
.classList.remove("hidden");

if(id=="products")
displayProducts();

if(id=="cart")
displayCart();

if(id=="orders")
displayOrders();

}


/* REGISTER */

function register(){

let u={

name:regName.value,
email:regEmail.value,
username:regUser.value,
password:regPass.value

};

if(!u.name ||
!u.email ||
!u.username ||
!u.password){

regMsg.innerHTML=
"Please fill all fields";

return;

}

users.push(u);

regMsg.innerHTML=
"Registration successful!";

}


/* LOGIN */

function login(){

let u=users.find(x=>

x.username==loginUser.value &&
x.password==loginPass.value

);

if(u){

currentUser=u.username;

loginMsg.innerHTML=
"Login successful!";

}else{

loginMsg.innerHTML=
"Invalid username or password";

}

}


/* DISPLAY PRODUCTS */

function displayProducts(){

let text=
search.value.toLowerCase();

let cat=category.value;
let occ=occasion.value;
let pr=price.value;

let result=products.filter(p=>

p.name.toLowerCase()
.includes(text) &&

(cat=="All" || p.category==cat) &&

(occ=="All" || p.occasion==occ) &&

(pr=="All" || p.price<=Number(pr))

);

productList.innerHTML="";

result.forEach(p=>{

productList.innerHTML+=`

<div class="card">

<div class="flower">🌷</div>

<h3>${p.name}</h3>

<p>${p.category}</p>

<p>${p.occasion}</p>

<p>${p.desc}</p>

<p class="price">
₹${p.price}
</p>

<p>Available</p>

<button class="btn"
onclick="addCart(${p.id})">

Add to Cart

</button>

</div>

`;

});

}


/* ADD TO CART */

function addCart(id){

let p=
products.find(x=>x.id==id);

cart.push(p);

alert(
p.name+" added to cart"
);

}


/* DISPLAY CART */

function displayCart(){

if(cart.length==0){

cartList.innerHTML=
"<p>Your cart is empty.</p>";

total.innerHTML="";

return;

}

let sum=0;

let html=`

<table>

<tr>
<th>Product</th>
<th>Price</th>
<th>Action</th>
</tr>
`;

cart.forEach((p,i)=>{

sum+=p.price;

html+=`

<tr>

<td>${p.name}</td>

<td>₹${p.price}</td>

<td>

<button class="btn"
onclick="removeCart(${i})">

Remove

</button>

</td>

</tr>

`;

});

html+="</table>";

cartList.innerHTML=html;

total.innerHTML=
"Total: ₹"+sum;

}


/* REMOVE ITEM */

function removeCart(i){

cart.splice(i,1);

displayCart();

}


/* CHECKOUT */

function checkout(){

if(!currentUser){

alert("Please login first");

show("login");

return;

}

if(cart.length==0){

alert("Cart is empty");

return;

}

if(!deliveryDate.value){

alert("Select delivery date");

return;

}

let amount=
cart.reduce(
(sum,p)=>sum+p.price,0
);

let order={

id:"ORD"+
Math.floor(10000+
Math.random()*90000),

user:currentUser,

items:cart.map(p=>p.name),

amount:amount,

delivery:deliveryDate.value,

time:deliveryTime.value,

status:"Order Confirmed"

};

orders.push(order);

cart=[];

alert(
"Order confirmed successfully!"
);

show("orders");

}


/* ORDER HISTORY */

function displayOrders(){

let data=
orders.filter(
o=>o.user==currentUser
);

if(data.length==0){

orderList.innerHTML=
"<p>No orders found.</p>";

return;

}

let html=`

<table>

<tr>
<th>Order ID</th>
<th>Products</th>
<th>Amount</th>
<th>Delivery</th>
<th>Status</th>
</tr>
`;

data.forEach(o=>{

html+=`

<tr>

<td>${o.id}</td>

<td>${o.items.join(", ")}</td>

<td>₹${o.amount}</td>

<td>${o.delivery}<br>${o.time}</td>

<td>${o.status}</td>

</tr>

`;

});

html+="</table>";

orderList.innerHTML=html;

}


/* ADMIN LOGIN */

function adminLogin(){

if(
adminUser.value=="admin" &&
adminPass.value=="admin123"
){

adminPanel.classList.remove(
"hidden"
);

displayAdmin();

}else{

alert("Invalid admin login");

}

}


/* ADD PRODUCT */

function addProduct(){

let p={

id:products.length+1,

name:newName.value,

category:newCategory.value,

occasion:newOccasion.value,

price:Number(newPrice.value),

desc:newDesc.value

};

if(!p.name || !p.price){

alert("Enter product details");

return;

}

products.push(p);

alert(
"Product added successfully"
);

displayAdmin();

displayProducts();

}


/* ADMIN MANAGEMENT */

function displayAdmin(){

let html=`

<table>

<tr>
<th>Product</th>
<th>Category</th>
<th>Price</th>
<th>Action</th>
</tr>
`;

products.forEach((p,i)=>{

html+=`

<tr>

<td>${p.name}</td>

<td>${p.category}</td>

<td>₹${p.price}</td>

<td>

<button class="btn"
onclick="deleteProduct(${i})">

Delete

</button>

</td>

</tr>

`;

});

html+="</table>";

adminProducts.innerHTML=html;


/* CUSTOMER INFO */

customers.innerHTML=
users.map(u=>`

<p>
<b>${u.name}</b> -
${u.email} -
${u.username}
</p>

`).join("");


/* ORDER SUMMARY */

adminOrders.innerHTML=
orders.map(o=>`

<p>

<b>${o.id}</b> -
${o.user} -
₹${o.amount} -
${o.status}

</p>

`).join("");

}


/* DELETE PRODUCT */

function deleteProduct(i){

products.splice(i,1);

displayAdmin();

displayProducts();

}
