<script>

/* DATA */

let users=[];

let products=[

{
id:1,
name:"Men Casual Shirt",
category:"Men",
sub:"Casual",
price:1299,
desc:"Comfortable cotton casual shirt"
},

{
id:2,
name:"Women Summer Dress",
category:"Women",
sub:"Casual",
price:1899,
desc:"Stylish summer dress"
},

{
id:3,
name:"Kids Party Wear",
category:"Children",
sub:"Formal",
price:999,
desc:"Elegant party wear for kids"
},

{
id:4,
name:"Running Sneakers",
category:"Footwear",
sub:"Sports",
price:2499,
desc:"Comfortable sports shoes"
},

{
id:5,
name:"Fashion Handbag",
category:"Accessories",
sub:"Casual",
price:1599,
desc:"Trendy women's handbag"
}

];

let cart=[];
let wishlist=[];
let orders=[];
let currentUser="";


/* NAVIGATION */

function show(id){

document.querySelectorAll("section")
.forEach(s=>s.classList.add("hidden"));

document.getElementById(id)
.classList.remove("hidden");

if(id=="products")
displayProducts();

if(id=="cart")
displayCart();

if(id=="wishlist")
displayWishlist();

if(id=="orders")
displayOrders();

}


/* REGISTER */

function register(){

let u={

name:regName.value,
email:regEmail.value,
username:regUser.value,
password:regPass.value

};

if(!u.name || !u.email ||
!u.username || !u.password){

regMsg.innerHTML=
"Please fill all fields";

return;
}

users.push(u);

regMsg.innerHTML=
"Registration successful!";

}


/* LOGIN */

function login(){

let u=users.find(x=>

x.username==loginUser.value &&
x.password==loginPass.value

);

if(u){

currentUser=u.username;

loginMsg.innerHTML=
"Login successful!";

}else{

loginMsg.innerHTML=
"Invalid username or password";

}

}


/* DISPLAY PRODUCTS */

function displayProducts(){

let text=
search.value.toLowerCase();

let cat=category.value;

let sub=subcategory.value;

let result=products.filter(p=>

p.name.toLowerCase().includes(text) &&

(cat=="All" || p.category==cat) &&

(sub=="All" || p.sub==sub)

);

productList.innerHTML="";

result.forEach(p=>{

productList.innerHTML+=`

<div class="product">

<div class="icon">🛍️</div>

<h3>${p.name}</h3>

<p>${p.category} - ${p.sub}</p>

<p>${p.desc}</p>

<p class="price">₹${p.price}</p>

<p>Available</p>

<button class="btn"
onclick="addCart(${p.id})">

Add to Cart

</button>

<button class="btn"
onclick="addWish(${p.id})">

♡ Wishlist

</button>

</div>

`;

});

}


/* CART */

function addCart(id){

let p=
products.find(x=>x.id==id);

cart.push(p);

alert(p.name+" added to cart");

}


function displayCart(){

if(cart.length==0){

cartList.innerHTML=
"<p>Cart is empty.</p>";

total.innerHTML="";

return;

}

let sum=0;

let html=`
<table>
<tr>
<th>Product</th>
<th>Price</th>
<th>Action</th>
</tr>`;

cart.forEach((p,i)=>{

sum+=p.price;

html+=`

<tr>

<td>${p.name}</td>

<td>₹${p.price}</td>

<td>
<button class="btn"
onclick="removeCart(${i})">
Remove
</button>
</td>

</tr>`;

});

html+="</table>";

cartList.innerHTML=html;

total.innerHTML=
"Total Amount: ₹"+sum;

}


function removeCart(i){

cart.splice(i,1);

displayCart();

}


/* WISHLIST */

function addWish(id){

let p=
products.find(x=>x.id==id);

wishlist.push(p);

alert(
p.name+" added to wishlist"
);

}


function displayWishlist(){

wishList.innerHTML="";

if(wishlist.length==0){

wishList.innerHTML=
"<p>Wishlist is empty.</p>";

return;

}

wishlist.forEach(p=>{

wishList.innerHTML+=`

<div class="product">

<h3>${p.name}</h3>

<p>₹${p.price}</p>

<button class="btn"
onclick="addCart(${p.id})">

Add to Cart

</button>

</div>`;

});

}


/* CHECKOUT */

function checkout(){

if(!currentUser){

alert("Please login first");

show("login");

return;

}

if(cart.length==0){

alert("Cart is empty");

return;

}

let amount=
cart.reduce(
(sum,p)=>sum+p.price,0
);

let order={

id:"ORD"+
Math.floor(10000+
Math.random()*90000),

user:currentUser,

items:cart.map(p=>p.name),

amount:amount,

status:"Order Placed"

};

orders.push(order);

cart=[];

alert("Order placed successfully!");

show("orders");

}


/* ORDERS */

function displayOrders(){

let data=
orders.filter(
o=>o.user==currentUser
);

if(data.length==0){

orderList.innerHTML=
"<p>No orders found.</p>";

return;

}

let html=`
<table>
<tr>
<th>Order ID</th>
<th>Items</th>
<th>Amount</th>
<th>Status</th>
</tr>`;

data.forEach(o=>{

html+=`

<tr>

<td>${o.id}</td>

<td>${o.items.join(", ")}</td>

<td>₹${o.amount}</td>

<td>${o.status}</td>

</tr>`;

});

html+="</table>";

orderList.innerHTML=html;

}


/* ADMIN LOGIN */

function adminLogin(){

if(adminUser.value=="admin" &&
adminPass.value=="admin123"){

adminPanel.classList.remove("hidden");

displayAdmin();

}else{

alert("Invalid admin login");

}

}


/* ADD PRODUCT */

function addProduct(){

let p={

id:products.length+1,

name:newName.value,

category:newCategory.value,

sub:newSub.value,

price:Number(newPrice.value),

desc:newDesc.value

};

if(!p.name || !p.price){

alert("Enter product details");

return;

}

products.push(p);

alert("Product added successfully");

displayAdmin();

displayProducts();

}


/* ADMIN MANAGEMENT */

function displayAdmin(){

let html=`
<table>
<tr>
<th>Product</th>
<th>Category</th>
<th>Price</th>
<th>Description</th>
<th>Action</th>
</tr>`;

products.forEach((p,i)=>{

html+=`

<tr>

<td>${p.name}</td>

<td>${p.category}</td>

<td>₹${p.price}</td>

<td>${p.desc}</td>

<td>

<button class="btn"
onclick="deleteProduct(${i})">

Delete

</button>

</td>

</tr>`;

});

html+="</table>";

adminProducts.innerHTML=html;

customers.innerHTML=users.map(u=>
`
<p>
<b>${u.name}</b> -
${u.email}
</p>
`
).join("");

}


/* DELETE PRODUCT */

function deleteProduct(i){

products.splice(i,1);

displayAdmin();

displayProducts();

}
