JSON.parse(localStorage.getItem("cakeUsers")) || [];

let orders =
JSON.parse(localStorage.getItem("cakeOrders")) || [];

let cart =
JSON.parse(localStorage.getItem("cakeCart")) || [];

let currentUser =
localStorage.getItem("cakeCurrentUser") || null;

let selectedCake = null;

let cakes =
JSON.parse(localStorage.getItem("cakes")) || [

{
id:1,
name:"Chocolate Cake",
category:"Birthday",
flavor:"Chocolate",
price:900,
image:"https://images.unsplash.com/photo-1578985545062-69928b1d9587",

},

{

flavor:"Vanilla",

},

{

flavor:"Red Velvet",

},

price:1000,
image:"https://images.unsplash.com/photo-1565958011703-44f9829ba187",
description:"Fresh strawberry cake with delicious strawberry cream."
}

];


function save(){

localStorage.setItem(
"cakeUsers",
JSON.stringify(users)
);

localStorage.setItem(
"cakeOrders",
JSON.stringify(orders)
);

localStorage.setItem(
"cakeCart",
JSON.stringify(cart)
);

localStorage.setItem(
"cakes",
JSON.stringify(cakes)
);

localStorage.setItem(
"cakeCurrentUser",
currentUser || ""
);

}


function show(id){

document
.querySelectorAll("section")
.forEach(s=>s.classList.add("hidden"));

document
.getElementById(id)
.classList.remove("hidden");

if(id=="shop")
displayCakes();

if(id=="cart")
displayCart();

if(id=="orders")
displayOrders();

}


function register(){

let name =
document.getElementById("regName").value;

let email =
document.getElementById("regEmail").value;

let username =
document.getElementById("regUser").value;

let password =
document.getElementById("regPass").value;

if(!name || !email || !username || !password){

regMsg.innerHTML =
"<span class='error'>Fill all fields</span>";

return;

}

if(users.some(u=>u.username==username)){

regMsg.innerHTML =
"<span class='error'>Username already exists</span>";

return;

}

users.push({
name,
email,
username,
password
});

save();

regMsg.innerHTML =
"<span class='success'>Registration successful!</span>";

}


function login(){

let username =
document.getElementById("loginUser").value;

let password =
document.getElementById("loginPass").value;

let user =
users.find(
u=>u.username==username &&
u.password==password
);

if(user){

currentUser=username;

save();

loginMsg.innerHTML =
"<span class='success'>Login successful!</span>";

}else{

loginMsg.innerHTML =
"<span class='error'>Invalid username or password</span>";

}

}


function displayCakes(){

let search =
document.getElementById("search").value
.toLowerCase();

let category =
document.getElementById("category").value;

let flavor =
document.getElementById("subcategory").value;

let price =
document.getElementById("price").value;

let result =
cakes.filter(c=>{

let searchMatch =
c.name.toLowerCase().includes(search);

let categoryMatch =
category=="All" ||
c.category==category;

let flavorMatch =
flavor=="All" ||
c.flavor==flavor;

let priceMatch =
price=="All" ||
c.price<=Number(price);

return searchMatch &&
categoryMatch &&
flavorMatch &&
priceMatch;

});


let output="";

result.forEach(c=>{

output+=`

<div class="cake">

<img src="${c.image}">

<div class="cake-content">

<h3>${c.name}</h3>

<p>${c.description}</p>

<p><b>Category:</b> ${c.category}</p>

<p><b>Flavor:</b> ${c.flavor}</p>

<h3>₹${c.price}</h3>

<button
class="btn"
onclick="viewCake(${c.id})">

View Description

</button>

</div>

</div>

`;

});

cakeList.innerHTML =
output ||
"<p>No cakes found.</p>";

}

selectedCake =
cakes.find(c=>c.id==id);

cakeDetails.innerHTML=`

<h2>${selectedCake.name}</h2>

<img

style="width:100%;max-width:400px;border-radius:10px">

<p>

${selectedCake.category}
</p>

<p>

${selectedCake.flavor}
</p>

<h3>₹${selectedCake.price}</h3>

<br>

<button
class="btn"
onclick="addToCart(${selectedCake.id})">

Add to Cart

</button>

<button
class="btn"
onclick="buyNow(${selectedCake.id})">

Buy Now

</button>

`;

show("details");

}


function addToCart(id){

if(!currentUser){

alert("Please login first");

show("login");

return;

}

let cake =
cakes.find(c=>c.id==id);

cart.push(cake);

save();

alert("Cake added to cart");

}


function buyNow(id){

if(!currentUser){

alert("Please login first");

show("login");

return;

}

selectedCake =
cakes.find(c=>c.id==id);

orderCake.innerHTML=`

<h3>${selectedCake.name}</h3>

<p>Base Price: ₹${selectedCake.price}</p>

`;

show("order");

}


function displayCart(){

if(cart.length==0){

cartList.innerHTML=
"<p>Your cart is empty.</p>";

return;

}

let output="<table>";

output+=`

<tr>
<th>Cake</th>
<th>Price</th>
<th>Action</th>
</tr>

`;

cart.forEach((c,index)=>{

output+=`

<tr>

<td>${c.name}</td>

<td>₹${c.price}</td>

<td>

<button
class="btn"
onclick="cartBuy(${index})">

Buy

</button>

<button
class="btn"
onclick="removeCart(${index})">

Remove

</button>

</td>

</tr>

`;

});

output+="</table>";

cartList.innerHTML=output;

}


function cartBuy(index){

selectedCake=cart[index];

orderCake.innerHTML=`

<h3>${selectedCake.name}</h3>

<p>Price: ₹${selectedCake.price}</p>

`;

show("order");

}


function removeCart(index){

cart.splice(index,1);

save();

displayCart();

}


function placeOrder(){

if(!currentUser){

alert("Please login");

return;

}

let addressValue =
document.getElementById("address").value;

let locationValue =
document.getElementById("location").value;

if(!addressValue || !locationValue){

alert("Enter shipping address");

return;

}

let size =
document.getElementById("cakeSize");

let sizePrice =
Number(size.value);

let total =
selectedCake.price + sizePrice;

let orderId =
"ORD"+Math.floor(
100000+Math.random()*900000
);

orders.push({

orderId,

user:currentUser,

cake:selectedCake.name,

size:size.options[size.selectedIndex].text,

message:
document.getElementById("cakeMessage").value,

address:addressValue,

location:locationValue,

payment:
document.getElementById("payment").value,

amount:total,

status:"Order Placed"

});

save();

orderMsg.innerHTML=`

<div class="card">

<h3 class="success">
Order Placed Successfully!
</h3>

<p>
<b>Order ID:</b> ${orderId}<br>
<b>Cake:</b> ${selectedCake.name}<br>
<b>Size:</b> ${size.options[size.selectedIndex].text}<br>
<b>Total:</b> ₹${total}<br>
<b>Status:</b> Order Placed
</p>

</div>

`;

}


function displayOrders(){

if(!currentUser){

orderList.innerHTML =
"<p class='error'>Please login first.</p>";

return;

}

let myOrders =
orders.filter(o=>o.user==currentUser);

if(myOrders.length==0){

orderList.innerHTML =
"No orders found.";

return;

}

let output="<table>";

output+=`

<tr>

<th>Order ID</th>
<th>Cake</th>
<th>Size</th>
<th>Amount</th>
<th>Shipping</th>
<th>Status</th>

</tr>

`;

myOrders.forEach(o=>{

output+=`

<tr>

<td>${o.orderId}</td>

<td>${o.cake}</td>

<td>${o.size}</td>

<td>₹${o.amount}</td>

<td>${o.location}</td>

<td>${o.status}</td>

</tr>

`;

});

output+="</table>";

orderList.innerHTML=output;

}

function addProduct(){

let name =
document.getElementById("productName").value;

let category =
document.getElementById("productCategory").value;

let flavor =
document.getElementById("productFlavor").value;

let price =
Number(document.getElementById("productPrice").value);

let image =
document.getElementById("productImage").value;

let description =
document.getElementById("productDescription").value;

if(!name || !price){

alert("Enter product details");

return;

}

cakes.push({

id:Date.now(),

name,

category,

flavor,

price,

image:image ||
"https://images.unsplash.com/photo-1578985545062-69928b1d9587",

description

});

save();

alert("Cake posted successfully!");

}

}

save();

}
