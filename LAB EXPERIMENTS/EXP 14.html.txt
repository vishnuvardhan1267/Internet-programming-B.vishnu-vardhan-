.querySelectorAll("section")
.forEach(s =>

);

.getElementById(id)
.classList.remove("hidden");


if(id=="services")
displayServices();

if(id=="book")
loadServices();

if(id=="dashboard")
displayDashboard();

/* ---------------- REGISTER ---------------- */

function register(){

let name =
document.getElementById("regName").value;

let email =
document.getElementById("regEmail").value;

let phone =
document.getElementById("regPhone").value;

let username =
document.getElementById("regUser").value;

let password =
document.getElementById("regPass").value;


if(!name ||
!email ||
!phone ||
!username ||
!password){

regMsg.innerHTML =
"<span class='error'>Fill all fields</span>";

return;

}


if(
users.some(
u => u.username == username
)
){

regMsg.innerHTML =
"<span class='error'>Username already exists</span>";

return;

}


users.push({

name,
email,
phone,
username,
password

});


save();


regMsg.innerHTML =
"<span class='success'>Registration successful!</span>";

}


/* ---------------- LOGIN ---------------- */

function login(){

document.getElementById("loginUser").value;

let password =
document.getElementById("loginPass").value;


let user =
users.find(
u =>
u.username == username &&
u.password == password
);


if(user){

currentUser =
username;

loginMsg.innerHTML =
"<span class='success'>Login successful!</span>";

}else{

loginMsg.innerHTML =
"<span class='error'>Invalid username or password</span>";

}

}

</button>

</div>

`;

});

output;

/* ---------------- AVAILABLE SLOTS ---------------- */

function showSlots(){

let date =
document.getElementById(
"appointmentDate"
).value;


if(!date) return;


let stylist =
Number(
document.getElementById(
"stylistSelect"
).value
);


let booked =
appointments
.filter(a =>
a.date==date &&
a.stylistId==stylist &&
a.status!="Cancelled"
)
.map(a=>a.time);


let output="";


slots.forEach(time=>{

if(!booked.includes(time)){

output+=`
<option value="${time}">
${time}
</option>
`;

}

});


if(output==""){

output=`
<option>
No slots available
</option>
`;

}


timeSlot.innerHTML =
output;

}


/* ---------------- BOOK APPOINTMENT ---------------- */

function bookAppointment(){

if(!currentUser){

alert("Please login first");

show("login");

return;

}


let serviceId =
Number(serviceSelect.value);

let stylistId =
Number(stylistSelect.value);

let date =
document.getElementById(
"appointmentDate"
).value;

let time =
document.getElementById(
"timeSlot"
).value;

let payment =
document.getElementById(
"paymentType"
).value;


if(!date ||
!time ||
time=="No slots available"){

alert(
"Select an available date and time"
);

return;

}


let service =
services.find(
s=>s.id==serviceId
);

let stylist =
stylists.find(
s=>s.id==stylistId
);


/* Payment calculation */

let amount=service.price;

let paymentStatus="Paid";


if(payment=="deposit"){

amount =
service.price * 0.30;

paymentStatus =
"Deposit Paid";

}


let appointmentId =
"APT" +
Math.floor(
100000 +
Math.random()*900000
);


appointments.push({

id:appointmentId,

username:currentUser,

serviceId:serviceId,

service:service.name,

stylistId:stylistId,

stylist:stylist.name,

date:date,

time:time,

amount:amount,

payment:payment,

paymentStatus:paymentStatus,

status:"Confirmed"

});


save();


bookingMsg.innerHTML=`

<div class="card">

<h3 class="success">
Appointment Confirmed!
</h3>

<p>

<b>Appointment ID:</b>
${appointmentId}<br>

<b>Service:</b>
${service.name}<br>

<b>Stylist:</b>
${stylist.name}<br>

<b>Date:</b>
${date}<br>

<b>Time:</b>
${time}<br>

<b>Payment:</b>
₹${amount}<br>

<b>Status:</b>
Confirmed

</p>

</div>

`;


showSlots();

}
