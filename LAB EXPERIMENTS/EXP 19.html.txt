<script>

/* USER DATA */

let users=[];
let agents=[];
let customers=[];
let currentUser="";

let policies=[

{
id:"POL001",
name:"Health Secure",
type:"Health",
premium:12000,
coverage:500000,
duration:"1 Year",
status:"Active"
},

{
id:"POL002",
name:"Auto Protect",
type:"Auto",
premium:9000,
coverage:300000,
duration:"1 Year",
status:"Active"
},

{
id:"POL003",
name:"Home Shield",
type:"Home",
premium:15000,
coverage:1000000,
duration:"1 Year",
status:"Active"
}

];

let userPolicies=[];


/* NAVIGATION */

function show(id){

document.querySelectorAll("section")
.forEach(s=>s.classList.add("hidden"));

document.getElementById(id)
.classList.remove("hidden");

if(id=="policies")
displayPolicies();

if(id=="dashboard")
displayMyPolicies();

}


/* REGISTER */

function register(){

let u={

name:regName.value,
email:regEmail.value,
username:regUser.value,
password:regPass.value

};

if(!u.name ||
!u.email ||
!u.username ||
!u.password){

regMsg.innerHTML=
"Please fill all fields";

return;

}

users.push(u);

regMsg.innerHTML=
"Registration successful!";

}


/* LOGIN */

function login(){

let u=users.find(x=>

x.username==loginUser.value &&
x.password==loginPass.value

);

if(u){

currentUser=u.username;

loginMsg.innerHTML=
"Login successful!";

}else{

loginMsg.innerHTML=
"Invalid username or password";

}

}


/* DISPLAY POLICIES */

function displayPolicies(){

let html="";

policies.forEach((p,i)=>{

html+=`

<div class="card">

<h2>${p.name}</h2>

<p>Type: ${p.type}</p>

<p>Coverage:
₹${p.coverage.toLocaleString()}
</p>

<p class="price">
Premium: ₹${p.premium}/year
</p>

<p>Duration: ${p.duration}</p>

<button class="btn"
onclick="buyPolicy(${i})">

Purchase Policy

</button>

</div>

`;

});

policyList.innerHTML=html;

}


/* BUY POLICY */

function buyPolicy(i){

if(!currentUser){

alert("Please login first");

show("login");

return;

}

let p=policies[i];

userPolicies.push({

user:currentUser,
policy:p,
status:"Active"

});

alert(
p.name+" purchased successfully!"
);

show("dashboard");

}


/* COMPARE */

function comparePolicies(){

let p1=
policies[policy1.value];

let p2=
policies[policy2.value];

compareResult.innerHTML=`

<table>

<tr>
<th>Feature</th>
<th>${p1.name}</th>
<th>${p2.name}</th>
</tr>

<tr>
<td>Type</td>
<td>${p1.type}</td>
<td>${p2.type}</td>
</tr>

<tr>
<td>Premium</td>
<td>₹${p1.premium}</td>
<td>₹${p2.premium}</td>
</tr>

<tr>
<td>Coverage</td>
<td>₹${p1.coverage}</td>
<td>₹${p2.coverage}</td>
</tr>

<tr>
<td>Duration</td>
<td>${p1.duration}</td>
<td>${p2.duration}</td>
</tr>

</table>

`;

}


/* MY POLICIES */

function displayMyPolicies(){

let data=
userPolicies.filter(
p=>p.user==currentUser
);

if(data.length==0){

myPolicies.innerHTML=
"<p>No active policies.</p>";

return;

}

let html=`

<table>

<tr>
<th>Policy No.</th>
<th>Policy</th>
<th>Premium</th>
<th>Coverage</th>
<th>Status</th>
</tr>
`;

data.forEach((x,i)=>{

html+=`

<tr>

<td>${x.policy.id}</td>

<td>${x.policy.name}</td>

<td>₹${x.policy.premium}</td>

<td>₹${x.policy.coverage}</td>

<td>${x.status}</td>

</tr>

`;

});

html+="</table>";

myPolicies.innerHTML=html;

}


/* PREMIUM PAYMENT */

function payPremium(){

if(!paymentPolicy.value){

paymentMsg.innerHTML=
"Enter policy number";

return;

}

paymentMsg.innerHTML=
"Premium payment successful for "+
paymentPolicy.value;

}


/* CLAIM */

function fileClaim(){

if(!claimPolicy.value ||
!claimReason.value){

claimMsg.innerHTML=
"Please enter all claim details";

return;

}

claimMsg.innerHTML=
"Claim submitted successfully. Claim ID: CLM"+
Math.floor(1000+Math.random()*9000);

}


/* AGENT REGISTER */

function registerAgent(){

let a={

name:agentName.value,
email:agentEmail.value,
username:agentUser.value,
password:agentPass.value,
status:"Pending Approval"

};

agents.push(a);

agentMsg.innerHTML=
"Agent registration submitted for admin approval.";

}


/* CREATE CUSTOMER */

function createCustomer(){

let c={

name:customerName.value,
email:customerEmail.value

};

customers.push(c);

customerMsg.innerHTML=
"Customer created successfully.";

}


/* SMS ALERT */

function sendSMS(){

if(!alertPolicy.value){

smsMsg.innerHTML=
"Enter policy number";

return;

}

smsMsg.innerHTML=
"SMS reminder sent to policyholder for "+
alertPolicy.value;

}


/* ADMIN LOGIN */

function adminLogin(){

if(
adminUser.value=="admin" &&
adminPass.value=="admin123"
){

adminPanel.classList.remove(
"hidden"
);

displayAdmin();

}else{

alert("Invalid admin login");

}

}


/* ADMIN DATA */

function displayAdmin(){

let html=`

<table>

<tr>
<th>Agent</th>
<th>Email</th>
<th>Status</th>
<th>Action</th>
</tr>
`;

agents.forEach((a,i)=>{

html+=`

<tr>

<td>${a.name}</td>

<td>${a.email}</td>

<td>${a.status}</td>

<td>

<button class="btn"
onclick="approveAgent(${i})">

Approve

</button>

</td>

</tr>

`;

});

html+="</table>";

agentList.innerHTML=html;


/* CUSTOMER DETAILS */

customerList.innerHTML=
customers.map(c=>`

<p>
<b>${c.name}</b> -
${c.email}
</p>

`).join("");

}


/* APPROVE AGENT */

function approveAgent(i){

agents[i].status="Approved";

displayAdmin();

alert("Agent approved successfully");

}
