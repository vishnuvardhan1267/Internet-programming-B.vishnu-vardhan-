<script>

let users = [];
let currentUser = null;
let bookings = [];

let trains = [
    {
        no: "101",
        name: "Chennai Express",
        source: "Chennai",
        destination: "Bangalore",
        date: "2026-08-25",
        time: "06:00 AM",
        sleeper: 20,
        ac: 10,
        sleeperFare: 450,
        acFare: 900
    },

    {
        no: "102",
        name: "Coimbatore Express",
        source: "Chennai",
        destination: "Coimbatore",
        date: "2026-08-25",
        time: "08:30 AM",
        sleeper: 20,
        ac: 10,
        sleeperFare: 500,
        acFare: 1000
    },

    {
        no: "103",
        name: "Madurai Express",
        source: "Chennai",
        destination: "Madurai",
        date: "2026-08-26",
        time: "09:00 PM",
        sleeper: 20,
        ac: 10,
        sleeperFare: 550,
        acFare: 1100
    }
];

let selectedTrain = null;


// SHOW SECTION
function showSection(id) {

    document.querySelectorAll("section").forEach(section => {
        section.classList.add("hidden");
    });

    document.getElementById(id).classList.remove("hidden");

    if(id === "history") {
        displayHistory();
    }
}


// REGISTER
function register() {

    let name = document.getElementById("regName").value;
    let email = document.getElementById("regEmail").value;
    let username = document.getElementById("regUser").value;
    let password = document.getElementById("regPass").value;

    if(!name || !email || !username || !password) {
        document.getElementById("regMsg").innerHTML =
            "<span class='danger'>Please fill all fields.</span>";
        return;
    }

    users.push({
        name,
        email,
        username,
        password
    });

    document.getElementById("regMsg").innerHTML =
        "<span class='success'>Registration successful!</span>";
}


// LOGIN
function login() {

    let username = document.getElementById("loginUser").value;
    let password = document.getElementById("loginPass").value;

    let user = users.find(
        u => u.username === username && u.password === password
    );

    if(user) {

        currentUser = user;

        document.getElementById("loginMsg").innerHTML =
            "<span class='success'>Login successful!</span>";

    } else {

        document.getElementById("loginMsg").innerHTML =
            "<span class='danger'>Invalid username or password.</span>";
    }
}


// SEARCH TRAINS
function searchTrains() {

    let source = document.getElementById("source").value.toLowerCase();
    let destination =
        document.getElementById("destination").value.toLowerCase();

    let date = document.getElementById("journeyDate").value;

    let result = trains.filter(train =>
        train.source.toLowerCase() === source &&
        train.destination.toLowerCase() === destination &&
        train.date === date
    );

    let output = "";

    if(result.length === 0) {

        output = "<p class='danger'>No trains available.</p>";

    } else {

        result.forEach(train => {

            output += `
            <div class="train">

                <h3>${train.name}</h3>

                <p>
                Train No: ${train.no}<br>
                Route: ${train.source} → ${train.destination}<br>
                Date: ${train.date}<br>
                Time: ${train.time}
                </p>

                <br>

                <b>Sleeper:</b>
                ₹${train.sleeperFare}
                | Available Seats: ${train.sleeper}

                <br>

                <b>AC:</b>
                ₹${train.acFare}
                | Available Seats: ${train.ac}

                <br><br>

                <button class="btn"
                    onclick="openBooking('${train.no}')">
                    Book Ticket
                </button>

            </div>
            `;
        });
    }

    document.getElementById("trainResults").innerHTML = output;
}


// OPEN BOOKING
function openBooking(trainNo) {

    if(!currentUser) {

        alert("Please login first.");
        showSection("login");
        return;
    }

    selectedTrain = trains.find(t => t.no === trainNo);

    document.getElementById("selectedTrain").innerHTML =
        `<b>${selectedTrain.name}</b> -
        ${selectedTrain.source} to
        ${selectedTrain.destination}`;

    showSection("booking");
}


// BOOK TICKET
function bookTicket() {

    if(!currentUser) {
        alert("Please login first.");
        return;
    }

    let classType =
        document.getElementById("classType").value;

    let count =
        parseInt(document.getElementById("passengers").value);

    let name =
        document.getElementById("passengerName").value;

    let age =
        document.getElementById("passengerAge").value;

    let gender =
        document.getElementById("gender").value;

    let payment =
        document.getElementById("payment").value;

    if(!name || !age || count < 1) {

        alert("Please enter passenger details.");
        return;
    }

    let available =
        classType === "Sleeper"
        ? selectedTrain.sleeper
        : selectedTrain.ac;

    if(count > available) {

        alert("Insufficient seats available.");
        return;
    }

    let fare =
        classType === "Sleeper"
        ? selectedTrain.sleeperFare
        : selectedTrain.acFare;

    let total = fare * count;

    // Reduce seats
    if(classType === "Sleeper")
        selectedTrain.sleeper -= count;
    else
        selectedTrain.ac -= count;

    let pnr =
        "PNR" + Math.floor(100000 + Math.random() * 900000);

    let booking = {

        pnr: pnr,

        username: currentUser.username,

        train: selectedTrain.name,

        trainNo: selectedTrain.no,

        source: selectedTrain.source,

        destination: selectedTrain.destination,

        date: selectedTrain.date,

        classType: classType,

        passengers: count,

        passengerName: name,

        age: age,

        gender: gender,

        payment: payment,

        fare: total,

        status: "CONFIRMED"
    };

    bookings.push(booking);

    document.getElementById("bookingResult").innerHTML = `

        <div class="train">

            <h3 class="success">
                Booking Successful!
            </h3>

            <p>
            <b>PNR:</b> ${pnr}<br>
            <b>Train:</b> ${selectedTrain.name}<br>
            <b>From:</b> ${selectedTrain.source}<br>
            <b>To:</b> ${selectedTrain.destination}<br>
            <b>Date:</b> ${selectedTrain.date}<br>
            <b>Class:</b> ${classType}<br>
            <b>Passenger:</b> ${name}<br>
            <b>Seats:</b> ${count}<br>
            <b>Total Fare:</b> ₹${total}<br>
            <b>Payment:</b> ${payment}<br>
            <b>Status:</b> CONFIRMED
            </p>

            <br>

            <button class="btn" onclick="window.print()">
                Print Ticket
            </button>

        </div>
    `;
}


// CHECK PNR
function checkPNR() {

    let pnr =
        document.getElementById("pnrInput").value;

    let booking =
        bookings.find(b => b.pnr === pnr);

    if(!booking) {

        document.getElementById("pnrResult").innerHTML =
            "<p class='danger'>PNR not found.</p>";

        return;
    }

    document.getElementById("pnrResult").innerHTML = `

        <div class="train">

        <h3>PNR Status</h3>

        <p>
        <b>PNR:</b> ${booking.pnr}<br>
        <b>Train:</b> ${booking.train}<br>
        <b>Journey:</b>
        ${booking.source} → ${booking.destination}<br>
        <b>Date:</b> ${booking.date}<br>
        <b>Class:</b> ${booking.classType}<br>
        <b>Status:</b>
        <span class="success">${booking.status}</span>
        </p>

        </div>
    `;
}


// BOOKING HISTORY
function displayHistory() {

    if(!currentUser) {

        document.getElementById("historyResult").innerHTML =
            "<p class='danger'>Please login first.</p>";

        return;
    }

    let myBookings =
        bookings.filter(b =>
            b.username === currentUser.username
        );

    if(myBookings.length === 0) {

        document.getElementById("historyResult").innerHTML =
            "No bookings found.";

        return;
    }

    let output = `
    <table>

        <tr>
            <th>PNR</th>
            <th>Train</th>
            <th>Date</th>
            <th>Class</th>
            <th>Fare</th>
            <th>Status</th>
            <th>Action</th>
        </tr>
    `;

    myBookings.forEach(b => {

        output += `

        <tr>

            <td>${b.pnr}</td>

            <td>${b.train}</td>

            <td>${b.date}</td>

            <td>${b.classType}</td>

            <td>₹${b.fare}</td>

            <td>${b.status}</td>

            <td>
                ${
                    b.status === "CONFIRMED"
                    ?
                    `<button class="btn"
                    onclick="cancelTicket('${b.pnr}')">
                    Cancel
                    </button>`
                    :
                    "Cancelled"
                }
            </td>

        </tr>
        `;
    });

    output += "</table>";

    document.getElementById("historyResult").innerHTML =
        output;
}


// CANCEL TICKET
function cancelTicket(pnr) {

    let booking =
        bookings.find(b => b.pnr === pnr);

    if(!booking) return;

    let confirmCancel =
        confirm("Are you sure you want to cancel this ticket?");

    if(!confirmCancel) return;

    // 80% refund
    let refund = booking.fare * 0.80;

    booking.status = "CANCELLED";

    alert(
        "Ticket cancelled successfully!\n" +
        "Refund Amount: ₹" + refund
    );

    displayHistory();
}
